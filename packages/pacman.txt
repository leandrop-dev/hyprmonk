# Packages from the official repos (all in Arch's extra or core: nothing from
# the AUR). `install.sh check` lists the missing ones, `install.sh packages`
# installs them. A bare Arch install (no desktop) needs every one: what a
# desktop would have brought (fonts, the PulseAudio layer) is here. The
# cursor and the wallpapers are downloaded by install.sh (not in the repos).

# Session
hyprland
hyprpm
uwsm

# Shell and theming
quickshell
matugen
qt6-imageformats   # WebP wallpapers
ttf-jetbrains-mono-nerd
tela-circle-icon-theme-purple
noto-fonts   # text in apps and notifications beyond JetBrains Mono
noto-fonts-emoji
xdg-user-dirs   # ~/Pictures and the rest, where the wallpapers live

# Apps the binds start
kitty
dolphin

# Keyboard and mouse shared with other computers (needs the portal below)
deskflow

# Root prompts: the shell is the polkit agent
polkit

# Locking when idle and before suspend (the shell draws the lock screen)
hypridle

# Secrets: KWallet, unlocked at login by hyprmonk-kwallet.service
kwallet
kwallet-pam
socat   # pam_kwallet_init talks through it

# Screenshots (hyprmonk screenshot) and screen sharing
grim
slurp
hyprpicker   # holds the screen still while picking a region or window
satty   # annotating before saving
wl-clipboard
libnotify   # notify-send
xdg-utils   # xdg-open: Open on the saved screenshot's notification
xdg-desktop-portal
xdg-desktop-portal-hyprland   # screen sharing, input capture (deskflow)
xdg-desktop-portal-gtk   # file pickers

# Audio, media and batteries
pipewire
pipewire-pulse   # apps speaking PulseAudio (pavucontrol, browsers)
pipewire-alsa   # apps speaking ALSA
wireplumber
pavucontrol   # the sound panel's Mixer
playerctl
upower
cava   # the centre card's spectrum

# Power profile (quick actions)
power-profiles-daemon

# Building hyprmonk and the hyprpm plugins
rustup
base-devel
cmake
meson
cpio
git
gtk4-layer-shell   # hymission's search box
