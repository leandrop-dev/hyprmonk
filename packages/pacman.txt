# Packages from the official repos (all in Arch's extra or core; the AUR ones
# are in aur.txt, the apps to pick from in apps.txt). `install.sh check` lists
# the missing ones, `install.sh packages` installs them. A bare CachyOS or Arch install (no desktop) needs
# every one: what a desktop would have brought (fonts, the PulseAudio layer,
# the network and Bluetooth services) is here. The cursor and the wallpapers
# are downloaded by install.sh (not in the repos).

# Session
hyprland
hyprpm
uwsm
sddm   # the login screen, with sddm-astronaut-theme (install.sh login)
qt6-svg   # the theme's icons
qt6-virtualkeyboard   # the theme's on-screen keyboard
qt6-multimedia-ffmpeg   # the theme's video backgrounds

# Shell and theming
quickshell
matugen
libvips
jq
qt6-imageformats   # WebP wallpapers
ttf-jetbrains-mono-nerd
noto-fonts   # text in apps and notifications beyond JetBrains Mono
noto-fonts-emoji
xdg-user-dirs   # ~/Pictures and the rest, where the wallpapers live
adw-gtk-theme   # GTK3 apps looking like GTK4 ones, in the tone's colours
breeze   # Qt apps' style (qt6ct)
kconfig   # kwriteconfig6: KDE apps' icons and style in kdeglobals

# Apps the binds and the shell start, and what opens files (install.sh
# defaults); the others are picked from apps.txt
kitty
nautilus   # files (Super+E); Show in folder on a screenshot
loupe      # images: Open on a screenshot

# The installer's menu
gum

# Files beyond the disk: phones, network shares, USB drives mounted as they
# are plugged in (hyprmonk-udiskie.service), video thumbnails
gvfs-mtp
gvfs-smb
udiskie
ffmpegthumbnailer

# Codecs for GStreamer apps; fonts for Chinese, Japanese and Korean text
gst-plugins-good
gst-plugins-bad
gst-plugins-ugly
gst-libav
noto-fonts-cjk

# Network and Bluetooth (install.sh system enables their services)
networkmanager
nm-connection-editor   # VPNs and the rest of a connection's settings
bluez
bluez-utils   # bluetoothctl

# Finding printers and other devices on the network (.local names), for
# Printing in apps.txt (install.sh system)
avahi
nss-mdns

# The terminal's shell: zsh with a prompt, smarter cd, fuzzy search,
# suggestions, highlighting and history search (config/zsh, install.sh shell)
zsh
starship
zoxide
fzf
zsh-autosuggestions
zsh-syntax-highlighting
zsh-history-substring-search
fastfetch   # the system at a glance as a terminal opens

# Clipboard history (hyprmonk-clipboard.service keeps it)
cliphist

# Root prompts: the shell is the polkit agent
polkit

# Locking when idle and before suspend (the shell draws the lock screen)
hypridle

# Secrets: GNOME Keyring, unlocked at login by SDDM's PAM file; SSH keys
# through gcr's agent (gcr-4 comes with it)
gnome-keyring

# Screenshots (hyprmonk screenshot) and screen sharing
grim
slurp
hyprpicker   # holds the screen still while picking a region or window
satty   # annotating before saving
wl-clipboard
libnotify   # notify-send
xdg-utils   # xdg-open: Open on the saved screenshot's notification
xdg-desktop-portal
xdg-desktop-portal-hyprland   # screen sharing, input capture (Deskflow)
xdg-desktop-portal-gtk   # file pickers

# Screen recording (hyprmonk record); it brings ffmpeg, for the notification's picture
gpu-screen-recorder

# Audio, media and batteries
pipewire
pipewire-pulse   # apps speaking PulseAudio (pavucontrol, browsers)
pipewire-alsa   # apps speaking ALSA
wireplumber
pavucontrol   # the sound panel's Mixer
playerctl
upower
cava   # the centre card's spectrum

# Power profile (quick actions)
power-profiles-daemon

# Building hyprmonk and the hyprpm plugins
rustup
base-devel
cmake
meson
cpio
git
gtk4-layer-shell   # hymission's search box
